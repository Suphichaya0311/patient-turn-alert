let countdown; let timeLeft = 0; let historyData = []; let alarmSound = new Audio("alarm.mp3"); function startTimer(hours, minutes, seconds){ timeLeft = hours*3600 + minutes*60 + seconds; countdown = setInterval(()=>{ timeLeft--; let h = Math.floor(timeLeft/3600); let m = Math.floor((timeLeft%3600)/60); let s = timeLeft%60; document.getElementById("countdown").innerText = `${h.toString().padStart(2,'0')}:${m.toString().padStart(2,'0')}:${s.toString().padStart(2,'0')}`; if(timeLeft <= 10){ document.getElementById("countdown").style.color="red"; } if(timeLeft <= 0){ clearInterval(countdown); triggerAlert(); } },1000); } function triggerAlert(){ document.getElementById("alertBox").style.display="flex"; alarmSound.play(); capturePhoto(); setTimeout(()=>{ document.getElementById("alertBox").style.display="none"; alarmSound.pause(); alarmSound.currentTime=0; },5000); } function capturePhoto(){ let now = new Date(); let timeString = now.toLocaleString(); let imgSrc = document.getElementById("cameraPreview").src; historyData.unshift({img:imgSrc,time:timeString}); renderGallery(); } function renderGallery(){ let gallery = document.getElementById("gallery"); gallery.innerHTML=""; historyData.forEach(data=>{ let card = document.createElement("div"); card.className="photo-card"; card.innerHTML=`
${data.time}
`; card.onclick=()=>openModal(data.img); gallery.appendChild(card); }); } function openModal(src){ document.getElementById("modalImg").src=src; document.getElementById("imageModal").style.display="flex"; } function closeModal(){ document.getElementById("imageModal").style.display="none"; }
